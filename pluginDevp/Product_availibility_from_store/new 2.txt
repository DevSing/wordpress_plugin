// Hook to track the last product update
add_action('save_post_product', 'wc_save_last_product_update_time', 10, 3);

function wc_save_last_product_update_time($post_id, $post, $update) {
    // Ensure this is a product post type
    if ($post->post_type !== 'product') {
        return;
    }

    // Update the option that stores the last product update time
    update_option('wc_last_product_update_time', current_time('mysql'));
}

// Add an admin menu item for the plugin
add_action('admin_menu', 'wc_last_update_admin_menu');

function wc_last_update_admin_menu() {
    add_menu_page(
        'Last Product Update',        // Page title
        'Last Product Update',        // Menu title
        'manage_options',             // Capability
        'wc-last-product-update',     // Menu slug
        'wc_display_last_product_update_page' // Function to display the page
    );
}

// Display the last product update time on the plugin page
function wc_display_last_product_update_page() {
    // Retrieve the last product update time
    $last_update_time = get_option('wc_last_product_update_time', __('No updates yet', 'woocommerce'));

    // Output the last update time
    echo '<div class="wrap">';
    echo '<h1>Last Product Update</h1>';
    echo '<p>The last product update happened on: <strong>' . esc_html($last_update_time) . '</strong></p>';
    echo '</div>';
}
